110QSharedPointer<TelegramImage> TelegramMsg::image(QualitySelector behavior, int size) const {
111 const QJsonArray&& array = rawJson()["photo"].toArray();
112 auto&& result = QSharedPointer<TelegramImage>::create();
113
114 switch (behavior) {
115 case QualitySelector::AroundSize:
116 {
117 int oldBestMath = std::numeric_limits<decltype(oldBestMath)>::max();
118 auto it = array.cbegin();
119
120 result->setRawJson(it->toObject());
121 it = std::next(it);
122
123 while (it != array.end() && std::abs(result->fileSize() - size) < oldBestMath) {
124 oldBestMath = std::abs(result->fileSize() - size);
125 result->setRawJson(it->toObject());
126 it = std::next(it);
127 }
128
129 return result;
130 }
131 case QualitySelector::BestOf: {
132 for (const auto& photo: array) {
133 auto photoObj = TelegramImage{photo.toObject()};
134 if (photoObj.fileSize() < size) {
135 result->setRawJson(photoObj.rawJson());
136 } else {
137 return result;
138 }
139 }
140
141 return result;
142 }
143 case QualitySelector::Best: {
144 if (array.size()) {
145 result->setRawJson(std::prev(array.end())->toObject());
146 }
147
148 return result;
149 }
150
151 case QualitySelector::Fast: {
152 if (array.size()) {
153 result->setRawJson(array.begin()->toObject());
154 }
155
156 return result;
157 }
158
159 break;
160 default:
161 break;
162 }
163
164 return result;
165}
